មនុស្សពីរនាក់យកច្បាប់ចម្លងនៃអគារដូចគ្នា ម្នាក់គូរស្លាបខាងកើត ហើយម្នាក់គូរស្លាបខាងលិច ហើយក្រោយមកច្បាប់ចម្លងទាំងនោះត្រូវក្លាយជាមួយ។ ប្រសិនបើពួកគេប៉ះពាល់វត្ថុផ្សេងគ្នា កម្មវិធីអាចធ្វើវាជំនួសមនុស្សបាន។

## អ្វីដែលធ្វើឱ្យវាដំណើរការ

ជញ្ជាំង ទ្វារ កម្រិត បន្ទប់ និងសំណុំលក្ខណៈសម្បត្តិនីមួយៗក្នុងឯកសារ IFC មាន GlobalId ដែលមិនដែលផ្លាស់ប្ដូរ។ ការបញ្ចូលរកវត្ថុតាមលេខសម្គាល់នោះ មិនមែនតាមលេខដែលឯកសារផ្ដល់ឱ្យ (`#123`) ទេ ព្រោះលេខផ្លាស់ប្ដូរនៅពេលកម្មវិធីរក្សាទុក។ ផ្នែកដែលព្យួរពីវត្ថុមួយ ហើយគ្មានលេខសម្គាល់ផ្ទាល់ខ្លួន (ការដាក់ទីតាំង រាង ឯកតា) ត្រូវបានប្រៀបធៀបតាមអ្វីដែលពួកវាមាន។

ច្បាប់ចម្លងទាំងពីរត្រូវតែជាកំណែ IFC ដូចគ្នា (2x3, 4 ឬ 4.3)។ ការបញ្ចូលរក្សាឯកសាររបស់អ្នកដូចដែលវាជា ជាមួយលំដាប់ និងលេខរបស់វា ហើយបន្ថែមអ្វីដែលថ្មី ដូច្នេះ diff របស់ git នៃការបញ្ចូលបង្ហាញតែអ្វីដែលការបញ្ចូលបានផ្លាស់ប្ដូរ។

## រៀបចំ

Git ត្រូវការពីរបន្ទាត់ម្ដងក្នុងមួយ repository៖

```
# .gitattributes
*.ifc merge=cadcraft

# ម្ដងគត់ នៅលើកុំព្យូទ័រនីមួយៗ
git config merge.cadcraft.driver "node /path/to/tools/cli/merge.mjs %O %A %B"
```

បើគ្មាន git ដំណើរការ `cadcraft merge <base> <ours> <theirs>`។ Fossil និង Mercurial យកឯកសារបីដូចគ្នាក្នុងការកំណត់ merge-tool របស់ពួកវាផ្ទាល់។ គ្មានអ្វីត្រូវបានបង្ហោះទេ; ទាំងអស់ដំណើរការនៅលើកុំព្យូទ័ររបស់អ្នក។

## ច្បាប់

- លក្ខណៈមួយដែលបានផ្លាស់ប្ដូរនៅម្ខាងតែប៉ុណ្ណោះ ត្រូវបានយក។ ការផ្លាស់ប្ដូរដូចគ្នានៅទាំងពីរខាង ត្រូវបានយកតែមួយដង។
- លក្ខណៈតែមួយដែលបានផ្លាស់ប្ដូរពីរវិធី (កម្រិតដូចគ្នាត្រូវបានប្ដូរឈ្មោះពីរវិធី) គឺជា **ជម្លោះ**៖ ខាងរបស់អ្នកនៅតែនៅ ហើយរបាយការណ៍រាយទាំងពីរ។ បើកលទ្ធផល ហើយកែវាដោយដៃ។
- បញ្ជីនៃអ្វីមួយ (ជញ្ជាំងនៃកម្រិតមួយ លក្ខណៈសម្បត្តិនៃសំណុំមួយ) បញ្ចូលជាសំណុំ៖ អ្វីដែលខាងណាមួយបានបន្ថែម ត្រូវបានរក្សា ហើយអ្វីដែលខាងណាមួយបានដកចេញ ត្រូវបានលុប។ មនុស្សពីរនាក់ម្នាក់ៗបន្ថែមជញ្ជាំងទៅកម្រិតដូចគ្នា មិនមានជម្លោះទេ។
- វត្ថុដែលបានបន្ថែមនៅខាងណាមួយ ត្រូវបានរក្សា ជាមួយរាង និងលក្ខណៈសម្បត្តិរបស់វា។ បានដកចេញនៅម្ខាង ហើយមិនបានប្ដូរនៅម្ខាងទៀត: ត្រូវបានលុប។ បានដកចេញនៅម្ខាង ហើយបានប្ដូរនៅម្ខាងទៀត: ត្រូវបានលុប ហើយរាយ។
- ត្រាពេលវេលា និងអ្នកនិពន្ធនៃឯកសារ និងប្រវត្តិម្ចាស់របស់វត្ថុនីមួយៗ នៅតែជារបស់អ្នក។

រូបរាងធរណីមាត្របញ្ចូលទាំងមូល៖ រាងរបស់ជញ្ជាំងមកពីម្ខាង មិនដែលលាយចំណុចម្ដងមួយៗទេ។ ឯកសារ IFC ដែលបានបង្ហាប់ (zipped) និង IFCX មិនត្រូវបានបញ្ចូលទេ។ បន្ទាប់ពីការបញ្ចូល បើកលទ្ធផលក្នុង BimCraft ហើយមើលផ្ទាំង {{ui:Problems}} មុនពេលអ្នករក្សាទុក។
